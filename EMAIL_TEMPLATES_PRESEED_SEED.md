## 📨 EMAIL GENERATOR SCRIPT (Python)

Save as: `generate_emails.py`

```python
import csv
from datetime import datetime, timedelta

# Load investor list
investors = [
    {
        'name': 'John Smith',
        'email': 'john@example.com',
        'firm': 'Tech Ventures',
        'type': 'micro_vc',
        'initial_date': '2026-08-29'
    },
    # Add more investors...
]

def generate_email(investor, template_type='initial', days_delay=0):
    """Generate personalized email from template"""
    
    name = investor['name']
    firm = investor['firm']
    email_type = investor['type']
    
    send_date = datetime.strptime(investor['initial_date'], '%Y-%m-%d') + timedelta(days=days_delay)
    
    if template_type == 'initial':
        if email_type == 'angel':
            subject = f"LANCAST: AI Construction Automation (Pre-Seed)"
            body = f"""Hi {name},

I saw you invest in construction tech on AngelList.

Quick intro: LANCAST is AI-powered construction automation. We automate 
contractor workflows using computer vision + IoT sensors.

Why now:
→ 1 paying client live (Honduras, recurring revenue)
→ Próspera ZEDE backing (Tim Draper ecosystem)
→ $50B+ TAM in construction (700K+ US contractors, <2% with AI)

Pre-seed round: $200K-$400K (Sept close, $8M cap)

Happy to share deck?

Best,
Mario
LANCAST LLC
+1 (954) 669-2788
mario@lancast.biz"""

        elif email_type == 'micro_vc':
            subject = f"Pre-Seed Opportunity: ConTech + Honduras"
            body = f"""Hi {name},

I'm Mario from LANCAST, building AI-powered construction automation. 
We're raising $200K-$400K pre-seed (close: Sept 1-7, 2026).

Why we're reaching out to {firm}:
→ You fund pre-seed ConTech startups
→ We have 1 paying client + operating model validated
→ We're in Próspera ZEDE (Tim Draper ecosystem)
→ Infinita Week (Sept 1-7) = perfect diligence window

Quick facts:
├─ Business model: B2B SaaS
├─ Traction: 1 paying client ($3K/month)
├─ Market: $50B+ TAM
├─ Valuation: $8M cap (SAFE)

Infinita Week is Sept 1-7 in La Ceiba. Would {firm} participate?

Deck attached. Happy to call this week.

Best,
Mario
LANCAST LLC
+1 (954) 669-2788"""
    
    elif template_type == 'followup_1':
        subject = "Re: LANCAST - Quick question on timing"
        body = f"""Hi {name},

Quick follow-up on my email from [DATE].

Pre-seed round is moving fast - interest from multiple investors on 
AngelList/OurCrowd. Aiming to close Sept 1 for Infinita Week.

If interested, best to move this week.

Deck attached. Let me know if you want to chat.

Best,
Mario"""
    
    elif template_type == 'followup_2':
        subject = "Last spot in LANCAST pre-seed (closes Sept 1)"
        body = f"""Hi {name},

Final follow-up. Pre-seed closes Sept 1, down to last spot.

If you've been considering it, now is the time. Call me: +1 (954) 669-2788

Best,
Mario"""
    
    return {
        'name': name,
        'email': investor['email'],
        'firm': firm,
        'subject': subject,
        'body': body,
        'send_date': send_date.strftime('%Y-%m-%d'),
        'template': template_type
    }

def export_to_csv(emails, filename='lancast_emails.csv'):
    """Export emails to CSV for Gmail/other platforms"""
    with open(filename, 'w', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=['send_date', 'name', 'email', 'firm', 'subject', 'body', 'template'])
        writer.writeheader()
        writer.writerows(emails)

# Generate email sequences
emails_to_send = []

for investor in investors:
    # Initial email
    emails_to_send.append(generate_email(investor, 'initial', days_delay=0))
    
    # Follow-up #1 (3 days later)
    emails_to_send.append(generate_email(investor, 'followup_1', days_delay=3))
    
    # Follow-up #2 (5 days later)
    emails_to_send.append(generate_email(investor, 'followup_2', days_delay=5))

# Export
export_to_csv(emails_to_send)
print(f"✓ Generated {len(emails_to_send)} emails")
print(f"✓ Exported to: lancast_emails.csv")
```
